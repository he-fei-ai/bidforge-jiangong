# 架构图 L3 叶子层文字垂直显示优化方案

> **优化目标**：将 L3（叶子层）节点文字从**横向排列**改为**垂直排列**，解决横向节点过多导致画布过宽的问题。  
> **适用场景**：青浦住宅项目质量管控体系等含 8+ 个 L3 节点的架构图。

---

## 一、问题分析

### 1.1 当前问题

**当前布局**（L3 横向）：
```
总宽度 ≈ 11 个 L3 节点 × (130 + 10) = 1540px + 留白 ≈ 1620px
```

**问题**：
- 页面横向溢出，需左右滚动
- 打印/导出 DOCX 时图片过宽，字体过小
- 屏幕展示需缩放才能看全

### 1.2 优化思路

**将 L3 节点从"横排"改为"竖排"**：

```
优化前（横排）：
┌──────┬──────┬──────┬──────┬──────┬──────┬──────┬──────┬──────┬──────┬──────┐
│ 公司 │ 质量 │ 项目 │ 技术 │ 质量 │ 专职 │ PC   │ 防水 │ 进场 │ BIM  │ 精密 │
```

## 二、布局算法调整

### 2.1 核心思路

**L3 节点改为竖向矩形**：
- 宽度：`60px`（窄）
- 高度：`200px`（高）
- 文字：`writing-mode: vertical-rl`（垂直排列）
- 间距：`8px`（更紧凑）

### 2.2 尺寸配置调整

```javascript
const CONFIG = {
  rootWidth: 240, rootHeight: 50,
  level2Width: 200, level2Height: 50,
  
  // L3 节点：改为竖向矩形
  level3Width: 60,           // 原 130 → 60
  level3Height: 200,         // 原 50 → 200
  
  level1Y: 0,
  level2Y: 120,
  level3Y: 240,
  level3Gap: 8,              // 原 10 → 8，更紧凑
  
  padding: 40,
  busOffset: 30,
  arrowSize: 5,
};
```

### 2.3 L3 文字垂直显示样式

```css
.node.level3 {
  /* 垂直文字排列 */
  writing-mode: vertical-rl;
  text-orientation: upright;   /* 保持汉字直立 */
  
  /* 字体 */
  font-size: 13px;
  letter-spacing: 2px;
  line-height: 1.4;
  
  /* 布局 */
  display: flex;
  align-items: center;
  justify-content: center;
  padding: 15px 8px;
  
  /* 背景 */
  background: #4a7cc7;
  color: #fff;
  border-radius: 4px;
  
  /* 文字换行 */
  white-space: normal;
  word-break: break-all;
}
```

**关键属性说明**：

| CSS 属性                    | 作用                       |
| --------------------------- | -------------------------- |
| `writing-mode: vertical-rl` | 文字从上到下、从右到左排列 |
| `text-orientation: upright` | 汉字保持直立（不旋转）     |
| `letter-spacing: 2px`       | 字间距，提升可读性         |
| `word-break: break-all`     | 长文本自动换行             |

### 3.2 React 组件（适配竖排）

```tsx
// components/QualitySystemChart.tsx
import React, { useMemo } from 'react';

export interface ArchNode {
  id: string;
  label: string;
  variant: 'root' | 'level2' | 'level3' | 'level3 highlight';
  children?: ArchNode[];
}

const DEFAULT_CONFIG = {
  rootWidth: 240, rootHeight: 50,
  level2Width: 200, level2Height: 50,
  level3Width: 60,
  level3Height: 200,
  level1Y: 0,
  level2Y: 120,
  level3Y: 240,
  level3Gap: 8,
  padding: 40,
  busOffset: 30,
  arrowSize: 5,
};

interface Props {
  root: ArchNode;
  config?: Partial<typeof DEFAULT_CONFIG>;
}

export const QualitySystemChart: React.FC<Props> = ({ root, config: userConfig }) => {
  const config = { ...DEFAULT_CONFIG, ...userConfig };

  const { nodeMap, canvasW, canvasH } = useMemo(() => {
    const nodeMap: Record<string, any> = {};
    let leafIndex = 0;

    const getSize = (v: string) => {
      if (v === 'root') return { w: config.rootWidth, h: config.rootHeight };
      if (v === 'level2') return { w: config.level2Width, h: config.level2Height };
      return { w: config.level3Width, h: config.level3Height };
    };

    const layoutNode = (node: ArchNode, depth: number, parentId: string | null): number => {
      const { w: width, h: height } = getSize(node.variant);

      let x: number;
      if (!node.children || node.children.length === 0) {
        const stride = config.level3Width + config.level3Gap;
        x = config.padding + width / 2 + leafIndex * stride;
        leafIndex++;
      } else {
        const childXs = node.children.map(c => layoutNode(c, depth + 1, node.id));
        x = (Math.min(...childXs) + Math.max(...childXs)) / 2;
      }

      const y = config.padding + height / 2 +
                (depth === 0 ? config.level1Y : depth === 1 ? config.level2Y : config.level3Y);

      nodeMap[node.id] = { ...node, x, y, width, height, depth, parentId };
      return x;
    };

    layoutNode(root, 0, null);

    const allNodes = Object.values(nodeMap) as any[];
    const canvasW = Math.max(...allNodes.map(n => n.x + n.width / 2)) + config.padding;
    const canvasH = Math.max(...allNodes.map(n => n.y + n.height / 2)) + config.padding;

    return { nodeMap, canvasW, canvasH };
  }, [root, config]);

  const allNodes = Object.values(nodeMap) as any[];

  const pathD = useMemo(() => {
    const childrenByParent: Record<string, any[]> = {};
    allNodes.forEach(n => {
      if (n.parentId) (childrenByParent[n.parentId] ||= []).push(n);
    });

    let d = '';
    Object.entries(childrenByParent).forEach(([parentId, children]) => {
      const parent = nodeMap[parentId];
      if (!parent) return;

      const startX = parent.x;
      const startY = parent.y + parent.height / 2;
      const busY = startY + config.busOffset;
      const leftX = Math.min(...children.map((c: any) => c.x));
      const rightX = Math.max(...children.map((c: any) => c.x));

      d += `M ${startX} ${startY} L ${startX} ${busY} `;
      if (children.length > 1) d += `M ${leftX} ${busY} L ${rightX} ${busY} `;
      children.forEach((c: any) => {
        const childTopY = c.y - c.height / 2;
        d += `M ${c.x} ${busY} L ${c.x} ${childTopY - config.arrowSize * 1.6} `;
      });
    });
    return d;
  }, [allNodes, nodeMap, config]);

  const arrows = useMemo(() => {
    const childrenByParent: Record<string, any[]> = {};
    allNodes.forEach(n => {
      if (n.parentId) (childrenByParent[n.parentId] ||= []).push(n);
    });
    return Object.values(childrenByParent).flat();
  }, [allNodes]);

  // ========== 节点样式 ==========
  const getNodeStyle = (variant: string): React.CSSProperties => {
    const base: React.CSSProperties = {
      position: 'absolute',
      display: 'flex',
      alignItems: 'center',
      justifyContent: 'center',
      borderRadius: 4,
      color: '#fff',
      textAlign: 'center',
      transition: 'transform .2s, box-shadow .2s',
      zIndex: 2,
      userSelect: 'none',
      boxSizing: 'border-box',
    };

    if (variant === 'root') {
      return {
        ...base,
        background: '#1f3a63',
        fontSize: 14,
        fontWeight: 600,
        padding: '0 12px',
        lineHeight: 1.3,
      };
    }

    if (variant === 'level2') {
      return {
        ...base,
        background: '#4a7cc7',
        fontSize: 13,
        fontWeight: 500,
        padding: '0 12px',
        lineHeight: 1.3,
      };
    }

    // L3：垂直文字
    const isHighlight = variant.includes('highlight');
    return {
      ...base,
      background: isHighlight ? '#f5b800' : '#4a7cc7',
      color: isHighlight ? '#1f3a63' : '#fff',
      fontSize: 13,
      fontWeight: isHighlight ? 600 : 500,
      padding: '15px 6px',
      
      // 关键：垂直文字
      writingMode: 'vertical-rl' as any,
      textOrientation: 'upright' as any,
      letterSpacing: 2,
      
      whiteSpace: 'normal',
      wordBreak: 'break-all',
      lineHeight: 1.4,
    };
  };

  return (
    <div style={{ position: 'relative', display: 'inline-block' }}>
      <div style={{
        fontSize: 18, fontWeight: 'bold', color: '#1f3a63',
        marginBottom: 30, letterSpacing: 1,
      }}>
        {root.label}
      </div>

      <div style={{ position: 'relative', width: canvasW, height: canvasH }}>
        <svg
          width={canvasW}
          height={canvasH}
          style={{ position: 'absolute', top: 0, left: 0, pointerEvents: 'none', zIndex: 1 }}
        >
          <path
            d={pathD}
            fill="none"
            stroke="#5a7ca8"
            strokeWidth={1.5}
            strokeLinecap="round"
            strokeLinejoin="round"
          />
          {arrows.map((c: any) => {
            const s = config.arrowSize;
            return (
              <polygon
                key={`arrow-${c.id}`}
                points={`${c.x},${c.y - c.height / 2} ${c.x - s},${c.y - c.height / 2 - s * 1.4} ${c.x + s},${c.y - c.height / 2 - s * 1.4}`}
                fill="#5a7ca8"
              />
            );
          })}
        </svg>

        {allNodes.map(n => (
          <div
            key={n.id}
            style={{
              ...getNodeStyle(n.variant),
              width: n.width,
              height: n.height,
              left: n.x - n.width / 2,
              top: n.y - n.height / 2,
            }}
            onMouseEnter={e => {
              (e.currentTarget as HTMLDivElement).style.transform = 'translateY(-2px)';
              (e.currentTarget as HTMLDivElement).style.boxShadow = '0 6px 18px rgba(30,80,160,.28)';
            }}
            onMouseLeave={e => {
              (e.currentTarget as HTMLDivElement).style.transform = '';
              (e.currentTarget as HTMLDivElement).style.boxShadow = '';
            }}
          >
            {n.label}
          </div>
        ))}
      </div>
    </div>
  );
};
```

```typescript
function getLayoutMode(l3Count: number): 'horizontal' | 'vertical' {
  // L3 节点 ≤ 6 个：横排
  // L3 节点 > 6 个：竖排
  return l3Count > 6 ? 'vertical' : 'horizontal';
}

const CONFIG = {
```

**结论**：  
将 L3 叶子层节点从**横向文字**改为**垂直文字**，可将画布宽度从 1620px 缩小至 750px（**减少 54%**），大幅改善 DOCX 导出、打印和屏幕展示效果。通过 `writing-mode: vertical-rl` + `text-orientation: upright` 实现中文竖排，同时保持高亮节点（黄色）的视觉突出。建议在软件中增加**自动判断节点数**的逻辑，L3 节点 > 6 个时自动启用竖排模式，≤ 6 个时保持横排。
